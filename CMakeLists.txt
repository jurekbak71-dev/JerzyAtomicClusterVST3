cmake_minimum_required(VERSION 3.25)
project(JerzyAtomicCluster VERSION 0.2.1 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
add_library(atomic_dsp INTERFACE)
target_include_directories(atomic_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(dsp_tests tests/DspTests.cpp)
target_link_libraries(dsp_tests PRIVATE atomic_dsp)
add_test(NAME dsp_regression COMMAND dsp_tests)
option(ATOMIC_BUILD_PLUGIN "Build the VST3 plugin" ON)
if(ATOMIC_BUILD_PLUGIN)
  set(VST3_SDK_ROOT "/workspace/dependencies/vst3sdk" CACHE PATH "Steinberg VST3 SDK root")
  if(NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Set VST3_SDK_ROOT to Steinberg VST3 SDK 3.8.1")
  endif()
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  if(CMAKE_CROSSCOMPILING)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_BUNDLE_FOR_WINDOWS OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${VST3_SDK_ROOT}" sdk)
  set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
  set(SDK_ROOT "${VST3_SDK_ROOT}")
  if(MINGW)
    # SDK 3.8.1's optional DataExchange helper uses std::aligned_alloc,
    # which MinGW's MSVCRT does not provide. This plugin does not expose
    # IDataExchangeHandler. Keep the SDK checkout intact and omit that helper.
    get_target_property(atomic_sdk_sources sdk SOURCES)
    list(FILTER atomic_sdk_sources EXCLUDE REGEX "/utility/dataexchange\\.cpp$")
    set_property(TARGET sdk PROPERTY SOURCES "${atomic_sdk_sources}")
  endif()
  smtg_add_vst3plugin(JerzyAtomicCluster src/Plugin.cpp src/Plugin.h src/Editor.cpp)
  target_link_libraries(JerzyAtomicCluster PRIVATE sdk atomic_dsp)
  target_compile_definitions(JerzyAtomicCluster PRIVATE ATOMIC_VERSION="${PROJECT_VERSION}")
  add_executable(render_audio tools/RenderAudio.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_link_libraries(render_audio PRIVATE sdk atomic_dsp ${CMAKE_DL_LIBS})
  target_compile_definitions(render_audio PRIVATE RELEASE=1)
  add_executable(processor_tests tests/ProcessorTests.cpp src/Plugin.cpp src/Editor.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_link_libraries(processor_tests PRIVATE sdk sdk_hosting atomic_dsp)
  target_compile_definitions(processor_tests PRIVATE RELEASE=1)
  target_compile_definitions(processor_tests PRIVATE ATOMIC_VERSION="${PROJECT_VERSION}")
  add_test(NAME processor_state COMMAND processor_tests)
  if(WIN32)
    add_executable(editor_smoke tests/EditorSmoke.cpp)
    target_link_libraries(editor_smoke PRIVATE sdk atomic_dsp gdi32 user32)
    target_compile_definitions(editor_smoke PRIVATE RELEASE=1)
    target_link_libraries(JerzyAtomicCluster PRIVATE gdi32 user32)
    target_link_libraries(processor_tests PRIVATE gdi32 user32)
    if(MINGW)
      target_link_options(JerzyAtomicCluster PRIVATE -static -static-libgcc -static-libstdc++)
      foreach(atomic_exe dsp_tests processor_tests validator editor_smoke moduleinfotool render_audio)
        target_link_options(${atomic_exe} PRIVATE -static -static-libgcc -static-libstdc++)
      endforeach()
      target_link_options(validator PRIVATE -municode)
    endif()
  endif()
endif()
